
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity bit4 is
    Port (
        a  : in  STD_LOGIC_VECTOR(3 downto 0);
        b  : in  STD_LOGIC_VECTOR(3 downto 0);
        op : in  STD_LOGIC_VECTOR(2 downto 0);
        c  : out STD_LOGIC_VECTOR(4 downto 0)
    );
end bit4;

architecture Behavioral of bit4 is
begin

    process(a, b, op)
    begin
        case op is

            when "000" =>
                c <= std_logic_vector(
                    unsigned('0' & a) + unsigned('0' & b)
                );

            when "001" =>
                c <= std_logic_vector(
                    unsigned('0' & a) - unsigned('0' & b)
                );

            when "010" =>
                c <= '0' & (a AND b);

            when "011" =>
                c <= '0' & (a OR b);

            when "100" =>
                c <= '0' & (a XOR b);

            when "101" =>
                c <= '0' & (a NAND b);

            when "110" =>
                c <= '0' & (a XNOR b);

            when "111" =>
                c <= '0' & a;

            when others =>
                c <= (others => '0');

        end case;
    end process;

end Behavioral;
