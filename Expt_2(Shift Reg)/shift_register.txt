
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity prr is
    Port (
        clr       : in  STD_LOGIC;
        sel       : in  STD_LOGIC_VECTOR(1 downto 0);
        shf_load  : in  STD_LOGIC;
        serial_in : in  STD_LOGIC;
        clk       : in  STD_LOGIC;
        d         : in  STD_LOGIC_VECTOR(3 downto 0);
        q         : out STD_LOGIC_VECTOR(3 downto 0)
    );
end prr;

architecture Behavioral of prr is
    signal temp : STD_LOGIC_VECTOR(3 downto 0) := "0000";
begin

    process(clk, clr)
    begin
        -- Asynchronous active-low clear
        if clr = '0' then
            temp <= "0000";

        elsif rising_edge(clk) then

            -- Parallel Load
            if shf_load = '0' then
                temp <= d;

            else
                case sel is

                    -- Shift Right
                    when "00" =>
                        temp <= serial_in & temp(3 downto 1);

                    -- Shift Left
                    when "01" =>
                        temp <= temp(2 downto 0) & serial_in;

                    -- Rotate Left
                    when "10" =>
                        temp <= temp(2 downto 0) & temp(3);

                    -- Rotate Right
                    when "11" =>
                        temp <= temp(0) & temp(3 downto 1);

                    when others =>
                        temp <= temp;

                end case;
            end if;
        end if;
    end process;

    q <= temp;

end Behavioral;
